% 双面打印分页（来自官方模板 preamble）
\ifx \twoSidePrint \undefined
  \def \ClearPageStyle{\clearpage}
\else
  \def \ClearPageStyle{%
    \clearpage
    \thispagestyle{empty}
    \quad
    \clearpage
  }
\fi
\newcounter{startpage}
\newcounter{endpage}
\setcounter{startpage}{1}
\setcounter{endpage}{1}
\newenvironment{PrintMode}{%
  \clearpage
  \setcounter{startpage}{\value{page}}%
}{%
  \clearpage
  \setcounter{endpage}{\value{page}}%
  \addtocounter{endpage}{\value{startpage}}%
  \ifodd \value{endpage}%
    \ClearPageStyle
    \ifx \twoSidePrint \undefined
    \else
      \addtocounter{page}{-1}%
    \fi
  \fi
}
\newcommand{\PrintModeInsert}[1]{\begin{PrintMode}#1\end{PrintMode}}
